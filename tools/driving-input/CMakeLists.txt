cmake_minimum_required(VERSION 3.20)
project(DiRT3VRDrivingInput LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Driving input capture requires Windows/MSVC.")
endif()
add_library(driving_input SHARED input.cpp)
target_compile_features(driving_input PRIVATE cxx_std_20)
target_compile_definitions(driving_input PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(driving_input PRIVATE dinput8 dxguid xinput9_1_0 ole32)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
    target_sources(driving_input PRIVATE exports.def)
endif()
# The test is kept in the development checkout only (not published).
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/axes_test.cpp")
    enable_testing()
    add_executable(driving_axes_test axes_test.cpp)
    target_compile_features(driving_axes_test PRIVATE cxx_std_20)
    target_compile_definitions(driving_axes_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(driving_axes_test PRIVATE dxguid)
    add_test(NAME driving_axes COMMAND driving_axes_test)
endif()
