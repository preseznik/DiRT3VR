function(add_proxy target)
  set(FRONTEND_SHADOW src/frontend_shadow.cpp src/frontend_shadow_state.cpp)
  add_library(${target} SHARED ${FRONTEND_SHADOW} src/proxy.cpp src/diagnostics.cpp src/scene_observer.cpp src/render_probe.cpp src/xr_runtime.cpp src/texture_presenter.cpp src/hud_capture.cpp src/hud_canvas.cpp src/hud_hooks.cpp src/hud_probe.cpp src/presentation_trace.cpp src/presentation_policy.cpp src/scene_projection.cpp src/scene_resolution.cpp src/game_vr.cpp src/intro_state.cpp src/camera_classifier.cpp src/background_play.cpp src/direct_practice.cpp src/screenshot.cpp src/seat_panel.cpp src/seat_input.cpp src/frame_timing.cpp src/gpu_usage.cpp src/performance_display.cpp src/seat_session.cpp src/desktop_seat.cpp src/anisotropic_filter.cpp src/cockpit_start.cpp "${CMAKE_BINARY_DIR}/exports.def")
  target_include_directories(${target} PRIVATE src "${CMAKE_BINARY_DIR}")
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
  target_compile_options(${target} PRIVATE /W4 /EHsc)
  # GDI+ draws only the seat menu panel; it loads when the panel is first drawn.
  target_link_libraries(${target} PRIVATE minhook bcrypt openxr_loader d3dcompiler.lib gdiplus.lib delayimp.lib dxguid.lib pdh.lib)
  target_link_options(${target} PRIVATE /DELAYLOAD:gdiplus.dll)
endfunction()
